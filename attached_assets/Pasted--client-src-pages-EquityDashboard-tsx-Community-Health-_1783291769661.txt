// client/src/pages/EquityDashboard.tsx
// Community Health Equity Dashboard — funder/commissioner audience
// Data: GET /api/benefits/sdoh-explorer/live + /api/benefits/sdoh-impact-chain/live
// RPLICE sidebar: GET /api/rplice/assessments + /api/rplice/reaim-scorecards
// Route: /equity-dashboard (public, no auth)

import { useState, useEffect } from "react";

// ─── Types ────────────────────────────────────────────────────────────────────

interface CountyData {
  fips: string;
  name: string;
  totalPop: number;
  eligible: number;
  enrolled: number;
  gap: number;
  gapRate: number;
  avgPoverty: number;
  avgBarrier: number;
  avgLimitedEnglish: number;
  avgNoBroadband: number;
  avgNoVehicle: number;
  avgUninsured: number;
  tractCount: number;
  highPovertyTracts: number;
  highBarrierTracts: number;
}

interface TractData {
  tractId: string;
  tractName: string;
  totalPop: number;
  medianIncome: number;
  povertyRate: number;
  limitedEnglishPct: number;
  noVehiclePct: number;
  noBroadbandPct: number;
  uninsuredRate: number;
  barrierIndex: number;
  eligiblePop: number;
  enrolledPop: number;
  gap: number;
}

interface ExplorerResponse {
  summary: {
    totalPopulation: number;
    totalEligible: number;
    totalEnrolled: number;
    totalGap: number;
    gapRate: number;
    totalTracts: number;
    highPovertyTracts: number;
    highBarrierTracts: number;
    noBroadbandTracts: number;
    noVehicleTracts: number;
    unclaimedBenefits: string;
  };
  counties: Record<string, CountyData>;
  topBarrierTracts: TractData[];
}

interface RpliceAssessment {
  id: string;
  programName: string;
  cfirScore?: number;
  domain?: string;
  status?: string;
}

interface ReaimScorecard {
  id: string;
  programName: string;
  reach?: number;
  effectiveness?: number;
  adoption?: number;
  implementation?: number;
  maintenance?: number;
}

// ─── County presets ───────────────────────────────────────────────────────────

const COUNTY_PRESETS = [
  { label: "Travis County", state: "48", county: "453", fips: "48453" },
  { label: "Williamson County", state: "48", county: "491", fips: "48491" },
  { label: "Hays County", state: "48", county: "209", fips: "48209" },
  { label: "Bastrop County", state: "48", county: "021", fips: "48021" },
];

// ─── Helpers ──────────────────────────────────────────────────────────────────

const fmt = (n: number) => n?.toLocaleString() ?? "—";
const pct = (n: number) => (n != null ? `${n.toFixed(1)}%` : "—");

function barrierLabel(index: number): { label: string; color: string } {
  if (index >= 30) return { label: "Critical", color: "#dc2626" };
  if (index >= 20) return { label: "High", color: "#ea580c" };
  if (index >= 10) return { label: "Moderate", color: "#ca8a04" };
  return { label: "Low", color: "#16a34a" };
}

function GapBar({ enrolled, gap }: { enrolled: number; gap: number }) {
  const total = enrolled + gap;
  const enrolledPct = total > 0 ? (enrolled / total) * 100 : 0;
  return (
    <div style={{ width: "100%", height: 10, borderRadius: 5, background: "#dc2626", overflow: "hidden" }}>
      <div style={{ width: `${enrolledPct}%`, height: "100%", background: "#10b981", borderRadius: 5 }} />
    </div>
  );
}

function MetricCard({
  label,
  value,
  sub,
  accent,
}: {
  label: string;
  value: string;
  sub?: string;
  accent?: string;
}) {
  return (
    <div style={{
      background: "rgba(255,255,255,0.04)",
      border: "1px solid rgba(255,255,255,0.08)",
      borderRadius: 12,
      padding: "20px 24px",
      display: "flex",
      flexDirection: "column",
      gap: 4,
    }}>
      <span style={{ fontSize: 11, letterSpacing: "0.08em", textTransform: "uppercase", color: "#94a3b8" }}>
        {label}
      </span>
      <span style={{ fontSize: 32, fontWeight: 700, color: accent ?? "#f1f5f9", lineHeight: 1.1 }}>
        {value}
      </span>
      {sub && <span style={{ fontSize: 13, color: "#64748b" }}>{sub}</span>}
    </div>
  );
}

function BarRow({
  label,
  value,
  max,
  color,
}: {
  label: string;
  value: number;
  max: number;
  color: string;
}) {
  const width = max > 0 ? Math.min((value / max) * 100, 100) : 0;
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 12, marginBottom: 10 }}>
      <span style={{ width: 140, fontSize: 12, color: "#94a3b8", flexShrink: 0 }}>{label}</span>
      <div style={{ flex: 1, height: 8, background: "rgba(255,255,255,0.06)", borderRadius: 4, overflow: "hidden" }}>
        <div style={{ width: `${width}%`, height: "100%", background: color, borderRadius: 4, transition: "width 0.6s ease" }} />
      </div>
      <span style={{ width: 44, fontSize: 12, color: "#e2e8f0", textAlign: "right" }}>{pct(value)}</span>
    </div>
  );
}

// ─── Main Component ───────────────────────────────────────────────────────────

export default function EquityDashboard() {
  const [selected, setSelected] = useState(COUNTY_PRESETS[0]);
  const [explorer, setExplorer] = useState<ExplorerResponse | null>(null);
  const [assessments, setAssessments] = useState<RpliceAssessment[]>([]);
  const [scorecards, setScorecards] = useState<ReaimScorecard[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    fetchData();
  }, [selected]);

  async function fetchData() {
    setLoading(true);
    setError(null);
    try {
      const [explorerRes, assessRes, reamRes] = await Promise.allSettled([
        fetch(`/api/benefits/sdoh-explorer/live?state=${selected.state}&counties=${selected.county}`),
        fetch(`/api/rplice/assessments`),
        fetch(`/api/rplice/reaim-scorecards`),
      ]);

      if (explorerRes.status === "fulfilled" && explorerRes.value.ok) {
        setExplorer(await explorerRes.value.json());
      } else {
        setError("Could not load equity data. Check API connection.");
      }

      if (assessRes.status === "fulfilled" && assessRes.value.ok) {
        const data = await assessRes.value.json();
        setAssessments(Array.isArray(data) ? data.slice(0, 4) : []);
      }

      if (reamRes.status === "fulfilled" && reamRes.value.ok) {
        const data = await reamRes.value.json();
        setScorecards(Array.isArray(data) ? data.slice(0, 3) : []);
      }
    } catch {
      setError("Network error. Verify RPLICE API is reachable.");
    } finally {
      setLoading(false);
    }
  }

  const county = explorer?.counties[selected.fips];
  const summary = explorer?.summary;
  const topTracts = explorer?.topBarrierTracts?.slice(0, 5) ?? [];

  return (
    <div style={{
      minHeight: "100vh",
      background: "#0a0f1a",
      color: "#e2e8f0",
      fontFamily: "'Inter', 'system-ui', sans-serif",
    }}>
      {/* Header */}
      <div style={{
        borderBottom: "1px solid rgba(255,255,255,0.06)",
        padding: "24px 40px",
        display: "flex",
        alignItems: "center",
        justifyContent: "space-between",
        flexWrap: "wrap",
        gap: 16,
      }}>
        <div>
          <div style={{ fontSize: 11, letterSpacing: "0.12em", textTransform: "uppercase", color: "#10b981", marginBottom: 4 }}>
            TCAF · RPLICE Community Intelligence
          </div>
          <h1 style={{ margin: 0, fontSize: 22, fontWeight: 700, color: "#f8fafc" }}>
            Community Health Equity Dashboard
          </h1>
        </div>

        {/* County selector */}
        <div style={{ display: "flex", gap: 8, flexWrap: "wrap" }}>
          {COUNTY_PRESETS.map((c) => (
            <button
              key={c.fips}
              onClick={() => setSelected(c)}
              style={{
                padding: "8px 16px",
                borderRadius: 8,
                border: selected.fips === c.fips
                  ? "1px solid #10b981"
                  : "1px solid rgba(255,255,255,0.1)",
                background: selected.fips === c.fips
                  ? "rgba(16,185,129,0.12)"
                  : "transparent",
                color: selected.fips === c.fips ? "#10b981" : "#94a3b8",
                fontSize: 13,
                cursor: "pointer",
                fontWeight: selected.fips === c.fips ? 600 : 400,
                transition: "all 0.2s",
              }}
            >
              {c.label}
            </button>
          ))}
        </div>
      </div>

      {/* Loading / Error */}
      {loading && (
        <div style={{ padding: "80px 40px", textAlign: "center", color: "#64748b" }}>
          Loading equity data for {selected.label}…
        </div>
      )}

      {error && (
        <div style={{ padding: "40px", margin: "40px", background: "rgba(220,38,38,0.1)", border: "1px solid rgba(220,38,38,0.3)", borderRadius: 12, color: "#fca5a5" }}>
          {error}
        </div>
      )}

      {!loading && !error && explorer && county && (
        <div style={{ padding: "32px 40px", maxWidth: 1400, margin: "0 auto" }}>

          {/* The headline number — the equity gap */}
          <div style={{
            background: "linear-gradient(135deg, rgba(220,38,38,0.15) 0%, rgba(10,15,26,0) 60%)",
            border: "1px solid rgba(220,38,38,0.2)",
            borderRadius: 16,
            padding: "32px 40px",
            marginBottom: 32,
            display: "flex",
            alignItems: "center",
            justifyContent: "space-between",
            flexWrap: "wrap",
            gap: 24,
          }}>
            <div>
              <div style={{ fontSize: 11, letterSpacing: "0.1em", textTransform: "uppercase", color: "#f87171", marginBottom: 8 }}>
                Benefits Access Gap · {county.name}
              </div>
              <div style={{ fontSize: 64, fontWeight: 800, color: "#fca5a5", lineHeight: 1 }}>
                {fmt(county.gap)}
              </div>
              <div style={{ fontSize: 16, color: "#94a3b8", marginTop: 8 }}>
                residents eligible for benefits — not enrolled
              </div>
            </div>
            <div style={{ textAlign: "right" }}>
              <div style={{ fontSize: 13, color: "#64748b", marginBottom: 12 }}>Enrollment rate</div>
              <div style={{ fontSize: 48, fontWeight: 700, color: "#10b981" }}>
                {pct(100 - county.gapRate)}
              </div>
              <GapBar enrolled={county.enrolled} gap={county.gap} />
              <div style={{ fontSize: 12, color: "#64748b", marginTop: 6 }}>
                {fmt(county.enrolled)} enrolled · {fmt(county.gap)} unreached
              </div>
            </div>
          </div>

          {/* Key metrics row */}
          <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fit, minmax(180px, 1fr))", gap: 16, marginBottom: 32 }}>
            <MetricCard label="Unclaimed Benefits" value={summary?.unclaimedBenefits ?? "—"} sub="estimated annual" accent="#f59e0b" />
            <MetricCard label="Avg Poverty Rate" value={pct(county.avgPoverty)} sub={`${county.highPovertyTracts} high-poverty tracts`} accent="#f87171" />
            <MetricCard label="Uninsured Rate" value={pct(county.avgUninsured)} sub="without health coverage" accent="#fb923c" />
            <MetricCard label="No Broadband" value={pct(county.avgNoBroadband)} sub={`${summary?.noBroadbandTracts} tracts affected`} accent="#a78bfa" />
            <MetricCard label="No Vehicle" value={pct(county.avgNoVehicle)} sub="transportation barrier" accent="#60a5fa" />
            <MetricCard label="Limited English" value={pct(county.avgLimitedEnglish)} sub="language barrier" accent="#34d399" />
          </div>

          {/* Two-column: barrier breakdown + top tracts */}
          <div style={{ display: "grid", gridTemplateColumns: "1fr 1.4fr", gap: 24, marginBottom: 32 }}>

            {/* Barrier breakdown */}
            <div style={{
              background: "rgba(255,255,255,0.03)",
              border: "1px solid rgba(255,255,255,0.07)",
              borderRadius: 14,
              padding: "28px 28px",
            }}>
              <div style={{ fontSize: 13, fontWeight: 600, color: "#94a3b8", marginBottom: 20, letterSpacing: "0.05em", textTransform: "uppercase" }}>
                Barrier Profile · {county.name}
              </div>
              <BarRow label="Poverty" value={county.avgPoverty} max={50} color="#f87171" />
              <BarRow label="Uninsured" value={county.avgUninsured} max={50} color="#fb923c" />
              <BarRow label="Limited English" value={county.avgLimitedEnglish} max={50} color="#34d399" />
              <BarRow label="No Broadband" value={county.avgNoBroadband} max={50} color="#a78bfa" />
              <BarRow label="No Vehicle" value={county.avgNoVehicle} max={50} color="#60a5fa" />
              <div style={{ marginTop: 20, paddingTop: 20, borderTop: "1px solid rgba(255,255,255,0.06)" }}>
                <div style={{ fontSize: 12, color: "#64748b", marginBottom: 4 }}>Composite Barrier Index</div>
                <div style={{ fontSize: 28, fontWeight: 700, color: "#f59e0b" }}>
                  {county.avgBarrier.toFixed(1)}
                </div>
                <div style={{ fontSize: 11, color: "#475569" }}>
                  (limitedEnglish×0.25) + (noVehicle×0.20) + (noBroadband×0.20) + (poverty×0.20)
                </div>
              </div>
            </div>

            {/* Top barrier tracts */}
            <div style={{
              background: "rgba(255,255,255,0.03)",
              border: "1px solid rgba(255,255,255,0.07)",
              borderRadius: 14,
              padding: "28px 28px",
            }}>
              <div style={{ fontSize: 13, fontWeight: 600, color: "#94a3b8", marginBottom: 20, letterSpacing: "0.05em", textTransform: "uppercase" }}>
                Highest-Barrier Census Tracts
              </div>
              {topTracts.length === 0 && (
                <div style={{ color: "#475569", fontSize: 13 }}>No tract data available.</div>
              )}
              {topTracts.map((tract, i) => {
                const { label, color } = barrierLabel(tract.barrierIndex);
                return (
                  <div key={tract.tractId} style={{
                    padding: "14px 0",
                    borderBottom: i < topTracts.length - 1 ? "1px solid rgba(255,255,255,0.05)" : "none",
                  }}>
                    <div style={{ display: "flex", justifyContent: "space-between", alignItems: "flex-start", marginBottom: 6 }}>
                      <div style={{ fontSize: 13, color: "#e2e8f0", fontWeight: 500, maxWidth: "70%" }}>
                        {tract.tractName}
                      </div>
                      <span style={{
                        fontSize: 11,
                        fontWeight: 600,
                        color,
                        background: `${color}18`,
                        border: `1px solid ${color}40`,
                        borderRadius: 4,
                        padding: "2px 8px",
                        whiteSpace: "nowrap",
                      }}>
                        {label} · {tract.barrierIndex.toFixed(1)}
                      </span>
                    </div>
                    <div style={{ display: "flex", gap: 16, fontSize: 12, color: "#64748b" }}>
                      <span>Poverty {pct(tract.povertyRate)}</span>
                      <span>Uninsured {pct(tract.uninsuredRate)}</span>
                      <span>Gap {fmt(tract.gap)}</span>
                    </div>
                    <div style={{ marginTop: 6 }}>
                      <GapBar enrolled={tract.enrolledPop} gap={tract.gap} />
                    </div>
                  </div>
                );
              })}
            </div>
          </div>

          {/* RPLICE sidebar — assessments + RE-AIM */}
          {(assessments.length > 0 || scorecards.length > 0) && (
            <div style={{
              background: "rgba(16,185,129,0.04)",
              border: "1px solid rgba(16,185,129,0.12)",
              borderRadius: 14,
              padding: "28px 32px",
              marginBottom: 32,
            }}>
              <div style={{ fontSize: 13, fontWeight: 600, color: "#10b981", marginBottom: 20, letterSpacing: "0.05em", textTransform: "uppercase" }}>
                RPLICE · Intervention Effectiveness
              </div>
              <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 32 }}>

                {/* CFIR Assessments */}
                {assessments.length > 0 && (
                  <div>
                    <div style={{ fontSize: 12, color: "#64748b", marginBottom: 12 }}>CFIR Program Assessments</div>
                    {assessments.map((a) => (
                      <div key={a.id} style={{
                        display: "flex",
                        justifyContent: "space-between",
                        alignItems: "center",
                        padding: "10px 0",
                        borderBottom: "1px solid rgba(255,255,255,0.04)",
                      }}>
                        <span style={{ fontSize: 13, color: "#cbd5e1" }}>{a.programName}</span>
                        {a.cfirScore != null && (
                          <span style={{
                            fontSize: 13,
                            fontWeight: 700,
                            color: a.cfirScore >= 70 ? "#10b981" : a.cfirScore >= 50 ? "#f59e0b" : "#f87171",
                          }}>
                            {a.cfirScore}
                          </span>
                        )}
                      </div>
                    ))}
                  </div>
                )}

                {/* RE-AIM Scorecards */}
                {scorecards.length > 0 && (
                  <div>
                    <div style={{ fontSize: 12, color: "#64748b", marginBottom: 12 }}>RE-AIM Scorecards</div>
                    {scorecards.map((s) => {
                      const scores = [
                        { label: "R", val: s.reach },
                        { label: "E", val: s.effectiveness },
                        { label: "A", val: s.adoption },
                        { label: "I", val: s.implementation },
                        { label: "M", val: s.maintenance },
                      ];
                      return (
                        <div key={s.id} style={{ marginBottom: 14 }}>
                          <div style={{ fontSize: 13, color: "#cbd5e1", marginBottom: 6 }}>{s.programName}</div>
                          <div style={{ display: "flex", gap: 8 }}>
                            {scores.map(({ label, val }) => (
                              <div key={label} style={{ textAlign: "center" }}>
                                <div style={{ fontSize: 10, color: "#475569" }}>{label}</div>
                                <div style={{ fontSize: 14, fontWeight: 700, color: val != null && val >= 70 ? "#10b981" : "#64748b" }}>
                                  {val != null ? val : "—"}
                                </div>
                              </div>
                            ))}
                          </div>
                        </div>
                      );
                    })}
                  </div>
                )}
              </div>
            </div>
          )}

          {/* Footer — data sources + print trigger */}
          <div style={{
            display: "flex",
            justifyContent: "space-between",
            alignItems: "center",
            paddingTop: 24,
            borderTop: "1px solid rgba(255,255,255,0.06)",
            flexWrap: "wrap",
            gap: 12,
          }}>
            <div style={{ fontSize: 11, color: "#475569", lineHeight: 1.6 }}>
              Data: U.S. Census Bureau ACS 5-Year · CDC PLACES · RPLICE Implementation Science Engine<br />
              County FIPS {selected.fips} · {county.tractCount} census tracts · Barrier index: AHC-HRSN + PRAPARE methodology
            </div>
            <button
              onClick={() => window.print()}
              style={{
                padding: "10px 20px",
                background: "rgba(16,185,129,0.1)",
                border: "1px solid rgba(16,185,129,0.3)",
                borderRadius: 8,
                color: "#10b981",
                fontSize: 13,
                cursor: "pointer",
                fontWeight: 600,
              }}
            >
              Export for briefing
            </button>
          </div>
        </div>
      )}
    </div>
  );
}
