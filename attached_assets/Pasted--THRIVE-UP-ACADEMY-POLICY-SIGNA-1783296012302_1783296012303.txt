// ── Signal types ─────────────────────────────────────────────

export interface PolicySignal {
  id: string;
  category: "workforce" | "equity" | "education" | "health" | "housing";
  finding: string;           // plain English, one sentence
  dataPoint: string;         // the number that anchors it

// ── Signal aggregator ─────────────────────────────────────────

export async function aggregateSignals(
  db: NodePgDatabase<any>,
  options: { countyFips?: string; daysBack?: number } = {}
): Promise<Record<string, unknown>> {
  const since = new Date();
  since.setDate(since.getDate() - (options.daysBack ?? 90));

  // 1. Workforce signals — credentials, placements, XP
  const [credentialCount] = await db
    .select({ count: count() })
    .from(certificates);

  const [placementCount] = await db
    .select({ count: count() })
    .from(outcomeTracking)
    .where(
      and(
        eq(outcomeTracking.metricName, "job_placement"),
        gte(outcomeTracking.measurementDate, since)
      )
    );

  const [learnerCount] = await db
    .select({ count: count() })
    .from(studentProgress);

  const [activeCount] = await db
    .select({ count: count() })
    .from(studentProgress)
    .where(gte(studentProgress.lastActiveDate ?? sql`now()`, since.toISOString().split("T")[0]));

  // 2. SDOH signals — household burden
  const [avgBurden] = await db
    .select({ avg: avg(householdSdohSnapshots.compositeBurdenScore) })
    .from(householdSdohSnapshots);

  const [highBurdenCount] = await db
    .select({ count: count() })
    .from(householdSdohSnapshots)
    .where(gte(householdSdohSnapshots.compositeBurdenScore, 35)); // ≥3.5 = high

  const [transportationBarrier] = await db
    .select({ count: count() })
    .from(householdSdohSnapshots)
    .where(eq(householdSdohSnapshots.transportationScore, 1));

  const [childcareBarrier] = await db
    .select({ count: count() })
    .from(householdSdohSnapshots)
    .where(eq(householdSdohSnapshots.childcareScore, 1));

  // 3. Household trajectory
  const [householdCount] = await db
    .select({ count: count() })
    .from(households);

  const [employedHouseholds] = await db
    .select({ count: count() })
    .from(householdOutcomes)
    .where(gte(householdOutcomes.employedMemberCount, 1));

  // 4. Pell eligibility flags
  const [pellCount] = await db
    .select({ count: count() })
    .from(outcomeTracking)
    .where(eq(outcomeTracking.metricName, "workforce_pell_eligible"));

  return {
    credentials: Number(credentialCount?.count ?? 0),
    placements90Days: Number(placementCount?.count ?? 0),
    totalLearners: Number(learnerCount?.count ?? 0),
    activeLearners90Days: Number(activeCount?.count ?? 0),
    avgSdohBurden: Number(avgBurden?.avg ?? 0) / 10,
    highBurdenScreenings: Number(highBurdenCount?.count ?? 0),
    transportationBarrierCount: Number(transportationBarrier?.count ?? 0),
    childcareBarrierCount: Number(childcareBarrier?.count ?? 0),
    totalHouseholds: Number(householdCount?.count ?? 0),
    employedHouseholds: Number(employedHouseholds?.count ?? 0),
    pellEligibleLearners: Number(pellCount?.count ?? 0),
  };
}

// ── AI narrative generator ────────────────────────────────────
// Calls Anthropic API to transform data into grant/policy language

export async function generatePolicyNarratives(
  signals: Record<string, unknown>,
  geography: string
): Promise<{
  grantNarrative: string;
  legislativeBrief: string;
  funderPitch: string;
  executiveSummary: string;
  recommendedAsk: string;
}> {
  const prompt = `You are a policy analyst and grant writer for TCAF (The Collaborative Advocate Foundation), a Texas 501(c)(3) serving marginalized communities in ${geography} through workforce development, SDOH navigation, and implementation science.

Here is live data from the ThriveUp Academy platform:

${JSON.stringify(signals, null, 2)}

Generate EXACTLY this JSON structure (no markdown, no explanation, just valid JSON):
{
  "grantNarrative": "A 3-sentence grant narrative paragraph suitable for a federal or foundation grant application. Lead with the equity gap, include specific numbers, end with the intervention and measurable outcome. Do not use 'we' — write in third person.",
  "legislativeBrief": "A 4-sentence legislative brief hook suitable for a county commissioner or state legislator. Open with the cost of inaction, include the intervention, cite the data, close with the specific ask.",
  "funderPitch": "A 2-sentence funder pitch hook for a program officer meeting. Lead with the most compelling data point, close with the ROI or leverage opportunity.",
  "executiveSummary": "A 5-sentence executive summary of current platform impact, suitable for a board report or annual report.",
  "recommendedAsk": "One specific, dollar-scoped funding ask based on the data, with justification in one sentence."
}`;

  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-6",
      max_tokens: 1500,
      messages: [{ role: "user", content: prompt }],
    }),
  });

  const data = await response.json();
  const text = data.content?.find((b: any) => b.type === "text")?.text ?? "{}";

  try {
    const clean = text.replace(/```json|```/g, "").trim();
    return JSON.parse(clean);
  } catch {
    return {
      grantNarrative: text.slice(0, 500),
      legislativeBrief: "",
      funderPitch: "",
      executiveSummary: "",
      recommendedAsk: "",
    };
  }
}

// ── Signal builder ────────────────────────────────────────────
// Converts raw aggregates into typed PolicySignal array

export function buildSignals(
  aggregates: Record<string, unknown>,
  narratives: Awaited<ReturnType<typeof generatePolicyNarratives>>,
  geography: string
): PolicySignal[] {
  const signals: PolicySignal[] = [];
  const now = new Date().toISOString();

  const creds = aggregates.credentials as number;
  const placements = aggregates.placements90Days as number;
  const learners = aggregates.totalLearners as number;
  const active = aggregates.activeLearners90Days as number;
  const highBurden = aggregates.highBurdenScreenings as number;
  const transport = aggregates.transportationBarrierCount as number;
  const childcare = aggregates.childcareBarrierCount as number;
  const pell = aggregates.pellEligibleLearners as number;
  const households = aggregates.totalHouseholds as number;
  const avgBurden = aggregates.avgSdohBurden as number;

  // Workforce signal
  if (creds > 0 || placements > 0) {
    signals.push({
      id: "workforce-credentials",
      category: "workforce",
      finding: `${creds} credentials earned; ${placements} job placements in the past 90 days`,
      dataPoint: `${creds} credentials · ${placements} placements`,
      geography,
      urgency: placements < creds * 0.3 ? "high" : "medium",
      grantNarrativeParagraph: narratives.grantNarrative,
      legislativeBriefHook: narratives.legislativeBrief,
      funderPitchHook: narratives.funderPitch,
      sourceData: { creds, placements, learners, active },
      generatedAt: now,
    });
  }

  // SDOH barrier signal
  if (highBurden > 0) {
    const urgency = avgBurden >= 4.5 ? "critical" : avgBurden >= 3 ? "high" : "medium";
    signals.push({
      id: "sdoh-barriers",
      category: "health",
      finding: `${highBurden} learner screenings show high/critical SDOH burden (avg ${avgBurden.toFixed(1)}/7)`,
      dataPoint: `${highBurden} high-burden · avg ${avgBurden.toFixed(1)}/7`,
      geography,
      urgency,
      grantNarrativeParagraph: `In ${geography}, ${highBurden} individuals served by TCAF's ThriveUp Academy report significant social determinants of health barriers, with an average burden score of ${avgBurden.toFixed(1)} out of 7 on the validated AHC-HRSN + PRAPARE instrument. Transportation (${transport} cases) and childcare (${childcare} cases) are the leading employment-blocking barriers, representing systemic gaps that workforce training alone cannot bridge. TCAF's integrated SDOH navigation model routes each participant to LifeBridge, ISSS, and partner agencies at the point of screening, reducing dropout before program completion.`,
      legislativeBriefHook: `Every untreated SDOH barrier costs Travis County an estimated $12,000–$18,000 annually in downstream healthcare and social services — ${highBurden} high-burden individuals in our current cohort represent a potential $${(highBurden * 15000).toLocaleString()} liability that preventive navigation can intercept. TCAF's SDOH intake model, deployed at the front door of workforce training, identifies and routes barriers before they become dropouts. The data: ${transport} individuals lack transportation, ${childcare} lack childcare — both are solvable with targeted wraparound funding. The ask: $${Math.round(highBurden * 800).toLocaleString()} to fund navigation services for the full identified cohort.`,
      funderPitchHook: `${highBurden} individuals in our active cohort are within reach of employment but blocked by transportation or childcare — not skill gaps. At $800 per person for SDOH navigation services, $${Math.round(highBurden * 800).toLocaleString()} turns ${highBurden} at-risk dropouts into job-ready completers with a documented credential.`,
      sourceData: { highBurden, transport, childcare, avgBurden },
      generatedAt: now,
    });
  }

  // Pell eligibility signal
  if (pell > 0) {
    signals.push({
      id: "pell-eligible",
      category: "education",
      finding: `${pell} learners have reached Workforce Pell eligibility threshold (3+ trade certifications)`,
      dataPoint: `${pell} Pell-eligible learners`,
      geography,
      urgency: "high",
      grantNarrativeParagraph: `${pell} participants in TCAF's ThriveUp Academy have completed three or more trade certifications, qualifying them for Workforce Pell Grant consideration under the WIOA reauthorization framework. This cohort represents a high-leverage investment opportunity: these individuals have demonstrated sustained engagement and multi-credential attainment, making them strong candidates for employer partnerships and apprenticeship pipelines. TCAF's implementation science engine tracks this threshold automatically, enabling proactive outreach and enrollment support at the moment of eligibility.`,
      legislativeBriefHook: `${pell} constituents in ${geography} have crossed the Workforce Pell eligibility threshold through TCAF's trade simulation platform — they are credential-ready and employment-adjacent. Workforce Pell expansion is federal policy; local implementation is where the gap exists. TCAF has the tracking infrastructure, the learner population, and the employer relationships to make ${geography} a pilot site for Workforce Pell deployment at scale. The ask: a letter of support from your office to accompany our federal application.`,
      funderPitchHook: `${pell} learners in our platform have earned 3+ trade credentials and qualify for Workforce Pell — federal dollars already authorized but not yet flowing to this community. Your investment bridges the implementation gap between federal eligibility and local access.`,
      sourceData: { pell, credentials: creds },
      generatedAt: now,
    });
  }

  // Household trajectory signal
  if (households > 0) {
    signals.push({
      id: "household-trajectory",
      category: "equity",
      finding: `${households} households tracked; platform treats poverty as a household phenomenon, not an individual one`,
      dataPoint: `${households} households · ${aggregates.employedHouseholds} with employed members`,
      geography,
      urgency: "medium",
      grantNarrativeParagraph: `TCAF's ThriveUp Academy operates a household unit model — the only workforce platform in ${geography} that tracks family-level economic trajectories rather than individual training completions. ${households} households are currently enrolled, with outcomes measured across employment, credential attainment, SDOH burden, and housing stability. This approach reflects the research consensus that poverty is a household phenomenon: individual credential attainment without household stability produces credential inflation, not economic mobility.`,
      legislativeBriefHook: `${geography} workforce programs measure individual placements. TCAF measures household economic trajectories — because a single placement in a household with three unaddressed SDOH barriers produces a 90-day retention rate, not a career. ${households} households are currently in TCAF's system. The data will show, within 18 months, whether household-level intervention outperforms individual-level programming on 90-day wage retention. Fund the comparison — the answer will shape county workforce policy for a decade.`,
      funderPitchHook: `${households} households enrolled, outcomes tracked at the family level — because employment of one member changes nothing if eviction threatens the household six months later. TCAF is the only organization in ${geography} with the data infrastructure to prove household-level ROI on workforce investment.`,
      sourceData: { households, employedHouseholds: aggregates.employedHouseholds },
      generatedAt: now,
    });
  }

  return signals;
}

export async function generateFullReport(
  db: NodePgDatabase<any>,
  geography: string,
  countyFips?: string
): Promise<PolicyReport> {
  const aggregates = await aggregateSignals(db, { countyFips });
  const narratives = await generatePolicyNarratives(aggregates, geography);
  const signals = buildSignals(aggregates, narratives, geography);

  return {
    geography,
    countyFips,
    signals,
    executiveSummary: narratives.executiveSummary,
    recommendedAsk: narratives.recommendedAsk,
    estimatedImpact: `${aggregates.totalLearners} learners · ${aggregates.credentials} credentials · ${aggregates.placements90Days} placements (90 days) · ${aggregates.totalHouseholds} households tracked`,
    generatedAt: new Date().toISOString(),
  };
}
