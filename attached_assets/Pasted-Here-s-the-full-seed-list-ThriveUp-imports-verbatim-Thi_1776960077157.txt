  scope: "state", state: code, county: county ?? null, area, programName,
  eligibility: extras.eligibility ?? {}, portalUrl, phone,
  applicationChannels: extras.applicationChannels ?? ["online","phone","in-person"],
  renewalCadence: extras.renewalCadence ?? "annual",
  annualDollarValueEstimate: extras.annualDollarValueEstimate,
  ...extras,
});

export const ALL_PROGRAMS: BenefitProgram[] = [...FEDERAL_PROGRAMS, ...STATE_PROGRAMS, ...COUNTY_PROGRAMS];
export function getProgramsForLocation(opts: { state?: string|null; county?: string|null; area?: BenefitAreaId|null } = {}): BenefitProgram[] {
  const state = opts.state ? opts.state.toUpperCase() : null;
  const county = opts.county ? opts.county.toLowerCase() : null;
  const area = opts.area || null;
  return ALL_PROGRAMS.filter(p => {
    if (area && p.area !== area) return false;
    if (p.scope === "federal") return true;
    if (p.scope === "state") return !state || p.state === state;
    if (p.scope === "county" || p.scope === "city") {
      if (!state || !county) return false;
      return p.state === state && (p.county || "").toLowerCase() === county;
    }
    return true;
  });
}
export function getProgramBySlug(slug: string): BenefitProgram | undefined {
  return ALL_PROGRAMS.find(p => p.slug === slug);
}

How both platforms stay in sync going forward
Rather than re-pasting the file every time one side adds a program, use the benefitProgram.updated event from the v2 contract:

Either platform adds/edits a program in its local catalog file
Fires benefitProgram.updated to RPLICE with the full BenefitProgram object as payload
Peer receives it, upserts by slug into its local catalog
Both sides now have the same record without a deploy on either
That's the mechanism that lets the catalog grow over time — every new county layer, every updated portal URL, every new state program propagates automatically.
