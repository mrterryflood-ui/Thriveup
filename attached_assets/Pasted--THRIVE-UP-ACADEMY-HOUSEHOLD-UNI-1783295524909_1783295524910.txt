export const householdRoleEnum = pgEnum("household_role", [
  "primary",      // whoever enrolled first — no hierarchy implied, just anchor
  "adult_member",
  "youth_member", // under 18
  "dependent",    // child not yet in platform
]);

export const consentDomainEnum = pgEnum("consent_domain", [
  "employment",
  "education",
  "housing",
  "food",
  "transportation",
  "childcare",
  "legal",
  "healthcare",
  "safety",
  "financial",
]);

export const consentLevelEnum = pgEnum("consent_level", [
  "private",      // only the member sees this
  "household",    // all household members see this
  "facilitator",  // member + assigned CHW/facilitator
  "program",      // member + program staff
]);

export const memberStatusEnum = pgEnum("member_status", [
  "active",
  "inactive",     // left household but record preserved
  "deceased",
  "incarcerated", // affects program eligibility
  "relocated",
]);

// ── Core Tables ───────────────────────────────────────────────

// households — first-class entity, no hierarchy
export const households = pgTable("households", {
  id: varchar("id", { length: 100 })
    .primaryKey()
    .default(sql`gen_random_uuid()`),
  householdCode: varchar("household_code", { length: 20 }).unique(), // human-readable e.g. HH-78653-0042
  zipCode: varchar("zip_code", { length: 10 }),
  countyFips: varchar("county_fips", { length: 10 }), // links to equity dashboard
  censustract: varchar("census_tract", { length: 20 }),
  createdAt: timestamp("created_at").defaultNow(),
  updatedAt: timestamp("updated_at").defaultNow(),
  notes: text("notes"),
  assignedFacilitatorId: varchar("assigned_facilitator_id", { length: 255 }), // CHW/promotora
  programCohort: varchar("program_cohort", { length: 100 }), // WIOA cohort, reentry cohort, etc.
});

// household_members — time-series membership, no static join
export const householdMembers = pgTable("household_members", {
  id: varchar("id", { length: 100 })
    .primaryKey()
    .default(sql`gen_random_uuid()`),
  householdId: varchar("household_id", { length: 100 })
    .notNull()
    .references(() => households.id, { onDelete: "cascade" }),
  userId: varchar("user_id", { length: 255 }), // null if dependent not in platform
  memberName: text("member_name").notNull(),
  role: householdRoleEnum("role").notNull().default("adult_member"),
  status: memberStatusEnum("status").notNull().default("active"),
  dateOfBirth: varchar("date_of_birth", { length: 20 }), // stored as string for privacy
  joinedHouseholdAt: timestamp("joined_household_at").defaultNow(),
  leftHouseholdAt: timestamp("left_household_at"), // null = still in household
  leftReason: text("left_reason"),
  isMinor: boolean("is_minor").default(false),
  hasVehicle: boolean("has_vehicle").default(false), // SDOH transport aggregation
  isEmployed: boolean("is_employed").default(false),
  employmentWage: integer("employment_wage"), // cents per hour
  notes: text("notes"),
});

// household_member_consent — domain-level privacy, explicit per member
export const householdMemberConsent = pgTable("household_member_consent", {
  id: varchar("id", { length: 100 })
    .primaryKey()
    .default(sql`gen_random_uuid()`),
  householdId: varchar("household_id", { length: 100 })
    .notNull()
    .references(() => households.id, { onDelete: "cascade" }),
  memberId: varchar("member_id", { length: 100 })
    .notNull()
    .references(() => householdMembers.id, { onDelete: "cascade" }),
  domain: consentDomainEnum("domain").notNull(),
  consentLevel: consentLevelEnum("consent_level").notNull().default("private"),
  consentedAt: timestamp("consented_at").defaultNow(),
  revokedAt: timestamp("revoked_at"), // null = still active
  consentedBy: varchar("consented_by", { length: 255 }), // userId of who gave consent
});

// household_sdoh_snapshots — member-level screening, aggregated at query time
export const householdSdohSnapshots = pgTable("household_sdoh_snapshots", {
  id: varchar("id", { length: 100 })
    .primaryKey()
    .default(sql`gen_random_uuid()`),
  householdId: varchar("household_id", { length: 100 })
    .notNull()
    .references(() => households.id, { onDelete: "cascade" }),
  memberId: varchar("member_id", { length: 100 })
    .notNull()
    .references(() => householdMembers.id, { onDelete: "cascade" }),
  screeningDate: timestamp("screening_date").defaultNow(),
  // Individual domain scores (0 = no barrier, 1 = barrier present)
  housingScore: integer("housing_score").default(0),
  foodScore: integer("food_score").default(0),
  transportationScore: integer("transportation_score").default(0),
  childcareScore: integer("childcare_score").default(0),
  legalScore: integer("legal_score").default(0),
  healthcareScore: integer("healthcare_score").default(0),
  safetyScore: integer("safety_score").default(0),
  // Weighted composite (0.0–7.0 per AHC-HRSN + PRAPARE)
  compositeBurdenScore: integer("composite_burden_score_x10").default(0), // stored ×10, e.g. 35 = 3.5
  burdenCategory: varchar("burden_category", { length: 20 }), // low/moderate/high/critical
  // Raw responses stored for audit
  rawResponses: jsonb("raw_responses"),
  // Partner routing output
  routedPartners: jsonb("routed_partners"), // [{domain, partner, resourceDescriptor}]
  screenedBy: varchar("screened_by", { length: 255 }), // userId of facilitator or "self"
});

// household_outcomes — household-level trajectory, separate from individual outcomeTracking
export const householdOutcomes = pgTable("household_outcomes", {
  id: varchar("id", { length: 100 })
    .primaryKey()
    .default(sql`gen_random_uuid()`),
  householdId: varchar("household_id", { length: 100 })
    .notNull()
    .references(() => households.id, { onDelete: "cascade" }),
  measuredAt: timestamp("measured_at").defaultNow(),
  // Employment
  employedMemberCount: integer("employed_member_count").default(0),
  totalHouseholdWageCents: integer("total_household_wage_cents").default(0),
  // Education & credentials
  activeLearnersCount: integer("active_learners_count").default(0),
  credentialsEarnedCount: integer("credentials_earned_count").default(0),
  // Stability indicators
  housingStable: boolean("housing_stable"),
  foodSecure: boolean("food_secure"),
  // SDOH composite across all active members
  householdBurdenScore: integer("household_burden_score_x10").default(0),
  // Flags
  pellEligibleFlag: boolean("pell_eligible_flag").default(false),
  wioa_enrolled: boolean("wioa_enrolled").default(false),
  reentryHousehold: boolean("reentry_household").default(false),
  notes: text("notes"),
  source: varchar("source", { length: 100 }).default("system"),
});

import { eq, and, isNull, desc, avg, sum, count } from "drizzle-orm";
import type { NodePgDatabase } from "drizzle-orm/node-postgres";

export interface HouseholdProfile {
  household: typeof households.$inferSelect;
  members: (typeof householdMembers.$inferSelect)[];
  latestSdoh: (typeof householdSdohSnapshots.$inferSelect)[];
  latestOutcome: typeof householdOutcomes.$inferSelect | null;
  aggregates: {
    totalMembers: number;
    activeMembers: number;
    employedCount: number;
    activeLearners: number;
    avgBurdenScore: number; // 0.0–7.0
    highestBurdenDomains: string[];
    combinedWagePerHour: number; // dollars
    credentialsEarned: number;
  };
  consentMap: Record<string, string>; // memberId → domain → consentLevel
}

export async function getHouseholdProfile(
  householdId: string,
  db: NodePgDatabase<any>,
  requestingUserId?: string // for consent filtering
): Promise<HouseholdProfile | null> {
  // 1. Household base
  const [household] = await db
    .select()
    .from(households)
    .where(eq(households.id, householdId));

  if (!household) return null;

  // 2. Active members (time-series: leftHouseholdAt IS NULL = still here)
  const members = await db
    .select()
    .from(householdMembers)
    .where(
      and(
        eq(householdMembers.householdId, householdId),
        isNull(householdMembers.leftHouseholdAt)
      )
    );

  // 3. Latest SDOH snapshot per member
  const latestSdoh = await db
    .select()
    .from(householdSdohSnapshots)
    .where(eq(householdSdohSnapshots.householdId, householdId))
    .orderBy(desc(householdSdohSnapshots.screeningDate))
    .limit(members.length * 2); // latest 2 per member max

  // 4. Latest outcome snapshot
  const [latestOutcome] = await db
    .select()
    .from(householdOutcomes)
    .where(eq(householdOutcomes.householdId, householdId))
    .orderBy(desc(householdOutcomes.measuredAt))
    .limit(1);

  // 5. Consent map (for facilitator view filtering)
  const consents = await db
    .select()
    .from(householdMemberConsent)
    .where(
      and(
        eq(householdMemberConsent.householdId, householdId),
        isNull(householdMemberConsent.revokedAt)
      )
    );

  const consentMap: Record<string, string> = {};
  for (const c of consents) {
    consentMap[`${c.memberId}:${c.domain}`] = c.consentLevel;
  }

  // 6. Compute aggregates
  const activeMembers = members.filter((m) => m.status === "active");
  const employedCount = activeMembers.filter((m) => m.isEmployed).length;
  const totalWageCents = activeMembers.reduce(
    (sum, m) => sum + (m.employmentWage ?? 0),
    0
  );

  // SDOH aggregation: average composite across latest snapshot per member
  const memberIds = new Set(members.map((m) => m.id));
  const seenMembers = new Set<string>();
  const relevantSdoh = latestSdoh.filter((s) => {
    if (!memberIds.has(s.memberId) || seenMembers.has(s.memberId)) return false;
    seenMembers.add(s.memberId);
    return true;
  });

  const avgBurden =
    relevantSdoh.length > 0
      ? relevantSdoh.reduce((s, r) => s + (r.compositeBurdenScore ?? 0), 0) /
        relevantSdoh.length /
        10
      : 0;

  // Identify highest burden domains across household
  const domainTotals: Record<string, number> = {
    housing: 0, food: 0, transportation: 0,
    childcare: 0, legal: 0, healthcare: 0, safety: 0,
  };
  for (const s of relevantSdoh) {
    domainTotals.housing += s.housingScore ?? 0;
    domainTotals.food += s.foodScore ?? 0;
    domainTotals.transportation += s.transportationScore ?? 0;
    domainTotals.childcare += s.childcareScore ?? 0;
    domainTotals.legal += s.legalScore ?? 0;
    domainTotals.healthcare += s.healthcareScore ?? 0;
    domainTotals.safety += s.safetyScore ?? 0;
  }
  const highestBurdenDomains = Object.entries(domainTotals)
    .filter(([, v]) => v > 0)
    .sort(([, a], [, b]) => b - a)
    .slice(0, 3)
    .map(([k]) => k);

  return {
    household,
    members,
    latestSdoh: relevantSdoh,
    latestOutcome: latestOutcome ?? null,
    aggregates: {
      totalMembers: members.length,
      activeMembers: activeMembers.length,
      employedCount,
      activeLearners: latestOutcome?.activeLearnersCount ?? 0,
      avgBurdenScore: Math.round(avgBurden * 10) / 10,
      highestBurdenDomains,
      combinedWagePerHour: totalWageCents / 100,
      credentialsEarned: latestOutcome?.credentialsEarnedCount ?? 0,
    },
    consentMap,
  };
}

// Find or create household for a userId
export async function findOrCreateHousehold(
  userId: string,
  memberName: string,
  db: NodePgDatabase<any>,
  options?: { zipCode?: string; countyFips?: string; censusTract?: string }
): Promise<{ householdId: string; memberId: string; isNew: boolean }> {
  // Check if user already belongs to a household
  const [existing] = await db
    .select({
      householdId: householdMembers.householdId,
      memberId: householdMembers.id,
    })
    .from(householdMembers)
    .where(
      and(
        eq(householdMembers.userId, userId),
        isNull(householdMembers.leftHouseholdAt)
      )
    );

  if (existing) {
    return { ...existing, isNew: false };
  }

  // Generate human-readable household code
  const zip = options?.zipCode ?? "00000";
  const rand = Math.floor(Math.random() * 9000) + 1000;
  const householdCode = `HH-${zip}-${rand}`;

  // Create household
  const [newHousehold] = await db
    .insert(households)
    .values({
      householdCode,
      zipCode: options?.zipCode,
      countyFips: options?.countyFips,
      censustrack: options?.censusTract,
    })
    .returning();

  // Add user as primary member
  const [newMember] = await db
    .insert(householdMembers)
    .values({
      householdId: newHousehold.id,
      userId,
      memberName,
      role: "primary",
      status: "active",
    })
    .returning();

  // Seed default consent — employment/education shared at household level
  await db.insert(householdMemberConsent).values([
    {
      householdId: newHousehold.id,
      memberId: newMember.id,
      domain: "employment",
      consentLevel: "household",
      consentedBy: userId,
    },
    {
      householdId: newHousehold.id,
      memberId: newMember.id,
      domain: "education",
      consentLevel: "household",
      consentedBy: userId,
    },
    // All other domains default to private per table definition
  ]);

  return {
    householdId: newHousehold.id,
    memberId: newMember.id,
    isNew: true,
  };
}

// Snapshot current household state into householdOutcomes
export async function snapshotHouseholdOutcomes(
  householdId: string,
  db: NodePgDatabase<any>
): Promise<void> {
  const profile = await getHouseholdProfile(householdId, db);
  if (!profile) return;

  const { aggregates, members } = profile;
  const reentryHousehold = members.some(
    (m) => m.notes?.toLowerCase().includes("reentry") || m.status === "incarcerated"
  );

  await db.insert(householdOutcomes).values({
    householdId,
    employedMemberCount: aggregates.employedCount,
    totalHouseholdWageCents: Math.round(aggregates.combinedWagePerHour * 100),
    activeLearnersCount: aggregates.activeLearners,
    credentialsEarnedCount: aggregates.credentialsEarned,
    householdBurdenScore: Math.round(aggregates.avgBurdenScore * 10),
    reentryHousehold,
    source: "system_snapshot",
  });
}

import express from "express";
import { z } from "zod";
import {
  getHouseholdProfile,
  findOrCreateHousehold,
  snapshotHouseholdOutcomes,
} from "./household-queries";

const router = express.Router();

// GET /api/households/:id — full profile with aggregates + consent map
router.get("/:id", async (req, res) => {
  try {
    const { id } = req.params;
    const requestingUserId = (req as any).user?.id;
    const profile = await getHouseholdProfile(id, req.app.locals.db, requestingUserId);
    if (!profile) return res.status(404).json({ error: "Household not found" });
    res.json(profile);
  } catch (err) {
    console.error("[household-routes] GET /:id", err);
    res.status(500).json({ error: "Failed to load household profile" });
  }
});

// POST /api/households — create or find household for a user
router.post("/", async (req, res) => {
  try {
    const schema = z.object({
      userId: z.string(),
      memberName: z.string(),
      zipCode: z.string().optional(),
      countyFips: z.string().optional(),
      censusTract: z.string().optional(),
    });
    const parsed = schema.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ error: parsed.error.flatten() });

    const result = await findOrCreateHousehold(
      parsed.data.userId,
      parsed.data.memberName,
      req.app.locals.db,
      {
        zipCode: parsed.data.zipCode,
        countyFips: parsed.data.countyFips,
        censusTract: parsed.data.censusTract,
      }
    );
    res.json(result);
  } catch (err) {
    console.error("[household-routes] POST /", err);
    res.status(500).json({ error: "Failed to create household" });
  }
});

// POST /api/households/:id/members — add a member to existing household
router.post("/:id/members", async (req, res) => {
  try {
    const schema = z.object({
      userId: z.string().optional(),
      memberName: z.string(),
      role: z.enum(["primary", "adult_member", "youth_member", "dependent"]).default("adult_member"),
      isMinor: z.boolean().default(false),
      hasVehicle: z.boolean().default(false),
      dateOfBirth: z.string().optional(),
      notes: z.string().optional(),
    });
    const parsed = schema.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ error: parsed.error.flatten() });

    const [member] = await req.app.locals.db
      .insert(householdMembers)
      .values({ householdId: req.params.id, ...parsed.data })
      .returning();

    res.json(member);
  } catch (err) {
    console.error("[household-routes] POST /:id/members", err);
    res.status(500).json({ error: "Failed to add member" });
  }
});

// PATCH /api/households/members/:memberId/status — handle moves, incarceration, etc.
router.patch("/members/:memberId/status", async (req, res) => {
  try {
    const schema = z.object({
      status: z.enum(["active", "inactive", "deceased", "incarcerated", "relocated"]),
      leftReason: z.string().optional(),
      leftHouseholdAt: z.string().optional(), // ISO date
    });
    const parsed = schema.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ error: parsed.error.flatten() });

    const [updated] = await req.app.locals.db
      .update(householdMembers)
      .set({
        status: parsed.data.status,
        leftReason: parsed.data.leftReason,
        leftHouseholdAt: parsed.data.leftHouseholdAt
          ? new Date(parsed.data.leftHouseholdAt)
          : parsed.data.status !== "active" ? new Date() : null,
      })
      .where(eq(householdMembers.id, req.params.memberId))
      .returning();

    res.json(updated);
  } catch (err) {
    console.error("[household-routes] PATCH /members/:memberId/status", err);
    res.status(500).json({ error: "Failed to update member status" });
  }
});

// POST /api/households/:id/sdoh — record SDOH screening for a member
router.post("/:id/sdoh", async (req, res) => {
  try {
    const schema = z.object({
      memberId: z.string(),
      housingScore: z.number().min(0).max(1).default(0),
      foodScore: z.number().min(0).max(1).default(0),
      transportationScore: z.number().min(0).max(1).default(0),
      childcareScore: z.number().min(0).max(1).default(0),
      legalScore: z.number().min(0).max(1).default(0),
      healthcareScore: z.number().min(0).max(1).default(0),
      safetyScore: z.number().min(0).max(1).default(0),
      compositeBurdenScore: z.number().min(0).max(70).default(0),
      burdenCategory: z.enum(["low", "moderate", "high", "critical"]).optional(),
      rawResponses: z.record(z.any()).optional(),
      routedPartners: z.array(z.any()).optional(),
      screenedBy: z.string().optional(),
    });
    const parsed = schema.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ error: parsed.error.flatten() });

    const [snapshot] = await req.app.locals.db
      .insert(householdSdohSnapshots)
      .values({
        householdId: req.params.id,
        ...parsed.data,
        compositeBurdenScore: Math.round(parsed.data.compositeBurdenScore),
      })
      .returning();

    // Auto-snapshot household outcomes whenever new SDOH data comes in
    await snapshotHouseholdOutcomes(req.params.id, req.app.locals.db).catch(
      (e) => console.error("[household-routes] snapshot failed:", e)
    );

    res.json(snapshot);
  } catch (err) {
    console.error("[household-routes] POST /:id/sdoh", err);
    res.status(500).json({ error: "Failed to record SDOH screening" });
  }
});

// PATCH /api/households/members/:memberId/consent — update consent for a domain
router.patch("/members/:memberId/consent", async (req, res) => {
  try {
    const schema = z.object({
      householdId: z.string(),
      domain: z.enum(["employment","education","housing","food","transportation",
                       "childcare","legal","healthcare","safety","financial"]),
      consentLevel: z.enum(["private","household","facilitator","program"]),
      consentedBy: z.string(),
    });
    const parsed = schema.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ error: parsed.error.flatten() });

    // Revoke any existing active consent for this member+domain
    await req.app.locals.db
      .update(householdMemberConsent)
      .set({ revokedAt: new Date() })
      .where(
        and(
          eq(householdMemberConsent.memberId, req.params.memberId),
          eq(householdMemberConsent.domain, parsed.data.domain),
          isNull(householdMemberConsent.revokedAt)
        )
      );

    // Insert new consent record
    const [consent] = await req.app.locals.db
      .insert(householdMemberConsent)
      .values({
        householdId: parsed.data.householdId,
        memberId: req.params.memberId,
        domain: parsed.data.domain,
        consentLevel: parsed.data.consentLevel,
        consentedBy: parsed.data.consentedBy,
      })
      .returning();

    res.json(consent);
  } catch (err) {
    console.error("[household-routes] PATCH /members/:memberId/consent", err);
    res.status(500).json({ error: "Failed to update consent" });
  }
});

// GET /api/households/:id/outcomes — trajectory over time
router.get("/:id/outcomes", async (req, res) => {
  try {
    const outcomes = await req.app.locals.db
      .select()
      .from(householdOutcomes)
      .where(eq(householdOutcomes.householdId, req.params.id))
      .orderBy(desc(householdOutcomes.measuredAt))
      .limit(24); // 24 snapshots = ~2 years at monthly cadence
    res.json(outcomes);
  } catch (err) {
    res.status(500).json({ error: "Failed to load outcomes" });
  }
});

// POST /api/households/:id/snapshot — manual trigger for outcome snapshot
router.post("/:id/snapshot", async (req, res) => {
  try {
    await snapshotHouseholdOutcomes(req.params.id, req.app.locals.db);
    res.json({ ok: true, snapshotted: new Date().toISOString() });
  } catch (err) {
    res.status(500).json({ error: "Snapshot failed" });
  }
});

export default router;

function BurdenBadge({ score }: { score: number }) {
  const category =
    score >= 5 ? "Critical" : score >= 3.5 ? "High" : score >= 2 ? "Moderate" : "Low";
  const color =
    score >= 5 ? "#dc2626" : score >= 3.5 ? "#ea580c" : score >= 2 ? "#ca8a04" : "#16a34a";
  return (
    <span style={{
      padding: "3px 10px", borderRadius: 6, fontSize: 12, fontWeight: 700,
      background: `${color}20`, color, border: `1px solid ${color}40`
    }}>
      {category} · {score.toFixed(1)}/7
    </span>
  );
}

function MemberCard({ member, sdoh, consentMap }: any) {
  const burdenScore = sdoh ? sdoh.compositeBurdenScore / 10 : null;
  const flaggedDomains = sdoh
    ? ["housing","food","transportation","childcare","legal","healthcare","safety"]
        .filter((d) => (sdoh as any)[`${d}Score`] > 0)
    : [];

  return (
    <div style={{
      background: "rgba(255,255,255,0.04)",
      border: "1px solid rgba(255,255,255,0.08)",
      borderRadius: 12,
      padding: "20px 24px",
      marginBottom: 12,
    }}>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "flex-start", marginBottom: 12 }}>
        <div>
          <div style={{ fontSize: 16, fontWeight: 600, color: "#f1f5f9" }}>{member.memberName}</div>
          <div style={{ fontSize: 12, color: "#64748b", marginTop: 2 }}>
            {member.role.replace("_", " ")} · {member.status}
            {member.isEmployed && " · Employed"}
            {member.hasVehicle && " · Has vehicle"}
          </div>
        </div>
        {burdenScore !== null && <BurdenBadge score={burdenScore} />}
      </div>

      {flaggedDomains.length > 0 && (
        <div>
          <div style={{ fontSize: 11, color: "#64748b", marginBottom: 8, letterSpacing: "0.06em", textTransform: "uppercase" }}>
            Barriers identified
          </div>
          <div style={{ display: "flex", flexWrap: "wrap", gap: 8 }}>
            {flaggedDomains.map((d) => (
              <span key={d} style={{
                padding: "4px 10px",
                borderRadius: 6,
                fontSize: 12,
                background: `${DOMAIN_COLORS[d]}18`,
                color: DOMAIN_COLORS[d],
                border: `1px solid ${DOMAIN_COLORS[d]}30`,
              }}>
                {d} → {DOMAIN_PARTNERS[d]}
              </span>
            ))}
          </div>
        </div>
      )}

      {member.isMinor && (
        <div style={{ marginTop: 8, fontSize: 12, color: "#a78bfa" }}>
          ⚠ Minor — parental consent required for program enrollment
        </div>
      )}
    </div>
  );
}
