export function computeEligibility(data: {
  annualIncome: number;
  householdSize: number;
  hasChildren?: boolean;
  isPregnant?: boolean;
  isDisabled?: boolean;
  isVeteran?: boolean;
  isSingleParent?: boolean;
  currentBenefits?: string[];
}): {
  eligible: string[];
  gaps: string[];
  estimatedAnnualValue: number;
  navigationGuides: typeof BENEFIT_NAVIGATION;
} {
  const income = data.annualIncome || 0;
  const hhSize = data.householdSize || 1;
  // 2024 FPL base: $15,060 for household of 1, +$5,380 per additional member
  const fpl = 15060 + (hhSize - 1) * 5380;
  const eligible: string[] = [];

  // ── Existing programs (verified correct) ─────────────────
  if (income <= fpl * 1.3) eligible.push("SNAP");
  if (income <= fpl * 1.38) eligible.push("Medicaid");
  if (data.hasChildren && income <= fpl * 2.0) eligible.push("CHIP");
  if (data.isPregnant || data.hasChildren) eligible.push("WIC");
  if (income <= fpl * 4.0) eligible.push("Marketplace");
  if (income > 0 && income <= fpl * 3.0) eligible.push("EITC");
  if (data.hasChildren && income <= fpl * 4.0) eligible.push("CTC");
  if (data.isDisabled) { eligible.push("SSI"); eligible.push("SSDI"); }

  // ── FIX: TANF (was completely missing) ───────────────────
  // Texas TANF: children in household, income ≤ 17% FPL for cash,
  // but federal screen is ≤ 200% FPL for program eligibility
  // Single parent flag increases likelihood but is not required
  if (
    data.hasChildren &&
    income <= fpl * 0.17  // Texas cash assistance threshold
  ) {
    eligible.push("TANF");
  }

  // ── NEW: CCDF Childcare Assistance ───────────────────────
  // Income ≤ 85% State Median Income (~$53,000 for family of 3 in TX)
  // Must be working, in school, or in job training
  if (data.hasChildren && income <= 53000) {
    eligible.push("CCDF");
  }

  // ── NEW: LIHEAP Utility Assistance ───────────────────────
  // Income ≤ 150% FPL
  if (income <= fpl * 1.5) {
    eligible.push("LIHEAP");
  }

  // ── NEW: Section 8 / Housing Choice Voucher ───────────────
  // Income ≤ 50% Area Median Income (AMI)
  // Austin AMI 2024: ~$110,300 for family of 4 → 50% = ~$55,150
  const austinAMI50pct = 55150 + (hhSize - 4) * 5000; // rough scaling
  if (income <= austinAMI50pct) {
    eligible.push("Section8");
  }

  // ── NEW: Veterans Benefits ────────────────────────────────
  // VA healthcare: any veteran with honorable discharge
  // No income limit for basic enrollment; Priority Group 1–8
  if (data.isVeteran) {
    eligible.push("VeteransBenefits");
  }

  const current = data.currentBenefits || [];
  const gaps = eligible.filter((b) => !current.includes(b));

  const estimatedAnnualValue = gaps.reduce((sum, b) => {
    return sum + (BENEFIT_NAVIGATION[b]?.estimatedAnnualValue ?? 0);
  }, 0);

  // Build navigation guides for gap programs only
  const navigationGuides = Object.fromEntries(
    gaps
      .filter((b) => BENEFIT_NAVIGATION[b])
      .map((b) => [b, BENEFIT_NAVIGATION[b]])
  ) as typeof BENEFIT_NAVIGATION;

  return { eligible, gaps, estimatedAnnualValue, navigationGuides };
}

// Simplified 8-domain risk/needs screen (LSI-R lite)
// Not a validated clinical instrument — intake triage only
// Flag for full LSI-R/ORAS assessment by case manager
interface RiskScreenResult {
  score: number;               // 0–16
  riskLevel: "low" | "medium" | "high";
  flaggedDomains: string[];
  recommendFullAssessment: boolean;
}

export function scoreRiskNeeds(answers: Record<string, boolean>): RiskScreenResult {
  const domains: Record<string, boolean> = {
    "Prior felony convictions (2+)": answers.priorFelonies ?? false,
    "Prior probation/parole violations": answers.priorViolations ?? false,
    "Unstable housing at release": answers.unstableHousing ?? false,
    "Unemployed at time of offense": answers.unemployedAtOffense ?? false,
    "Less than HS diploma/GED": answers.noHsDiploma ?? false,
    "History of substance use": answers.substanceHistory ?? false,
    "Family/social network with criminal history": answers.antisocialNetwork ?? false,
    "Self-reported mental health needs": answers.mentalHealthNeeds ?? false,
  };

  const flaggedDomains = Object.entries(domains)
    .filter(([, v]) => v)
    .map(([k]) => k);

  const score = flaggedDomains.length * 2; // 0–16
  const riskLevel =
    score >= 10 ? "high" : score >= 6 ? "medium" : "low";

  return {
    score,
    riskLevel,
    flaggedDomains,
    recommendFullAssessment: score >= 6,
  };
}

// Ban-the-box employer match — by location string partial match
export async function matchFairChanceEmployers(
  location?: string
): Promise<Array<{
  employer: typeof employerPartners.$inferSelect;
  openPostings: typeof jobPostings.$inferSelect[];
}>> {
  const employers = await db
    .select()
    .from(employerPartners)
    .where(
      and(
        eq(employerPartners.banTheBox, true),
        eq(employerPartners.partnershipStatus, "active")
      )
    );

  const results = await Promise.all(
    employers.map(async (employer) => {
      const openPostings = await db
        .select()
        .from(jobPostings)
        .where(
          and(
            eq(jobPostings.employerId, employer.id),
            eq(jobPostings.status, "open"),
            eq(jobPostings.barrierFriendly, true)
          )
        );
      return { employer, openPostings };
    })
  );

  // Filter to employers with at least one open posting
  return results.filter((r) => r.openPostings.length > 0);
}

// Enhanced participant creation handler
// Replaces the body of POST /api/intake/participants
export async function createParticipantEnhanced(
  participantData: typeof participantProfiles.$inferInsert,
  options: {
    riskAnswers?: Record<string, boolean>;
    zipCode?: string;
    countyFips?: string;
  } = {}
): Promise<{
  profile: typeof participantProfiles.$inferSelect;
  householdId: string;
  benefitsScreening: ReturnType<typeof computeEligibility> | null;
  employerMatches: Awaited<ReturnType<typeof matchFairChanceEmployers>>;
  riskScreen: RiskScreenResult | null;
}> {
  // 1. Create participant profile (existing behavior)
  const [profile] = await db
    .insert(participantProfiles)
    .values(participantData)
    .returning();

  // Run all side effects in parallel — failures are isolated
  const [householdResult, benefitsResult, employerResult, riskResult] =
    await Promise.allSettled([

      // 2. Create/link household record
      (async () => {
        const { householdId } = await findOrCreateHousehold(
          participantData.userId ?? profile.id,
          participantData.firstName + " " + participantData.lastName,
          db,
          {
            zipCode: options.zipCode,
            countyFips: options.countyFips ?? "48453", // default Travis County
          }
        );
        // Tag household as reentry
        await db
          .update(households)
          .set({ programCohort: "reentry" })
          .where(eq(households.id, householdId));
        return householdId;
      })(),

      // 3. Auto-run benefits screening
      (async () => {
        // Use what we know from participant profile
        const annualIncome = 0; // intake default — no income at release
        const hasChildren =
          (participantData as any).dependentsCount > 0 ?? false;
        return computeEligibility({
          annualIncome,
          householdSize: 1 + ((participantData as any).dependentsCount ?? 0),
          hasChildren,
          isVeteran: (participantData as any).isVeteran ?? false,
          currentBenefits: [],
        });
      })(),

      // 4. Query fair-chance employer matches
      matchFairChanceEmployers(options.zipCode),

      // 5. Score risk needs if answers provided
      (async () => {
        if (!options.riskAnswers) return null;
        return scoreRiskNeeds(options.riskAnswers);
      })(),
    ]);

  return {
    profile,
    householdId:
      householdResult.status === "fulfilled" ? householdResult.value : "unlinked",
    benefitsScreening:
      benefitsResult.status === "fulfilled" ? benefitsResult.value : null,
    employerMatches:
      employerResult.status === "fulfilled" ? employerResult.value : [],
    riskScreen:
      riskResult.status === "fulfilled" ? riskResult.value : null,
  };
}

// GET /api/workforce/match/:userId
// Returns fair-chance employers + open postings matched to earned credentials
export async function getEmployerMatches(userId: string): Promise<{
  userId: string;
  credentialsEarned: string[];
  fairChanceMatches: Array<{
    employer: typeof employerPartners.$inferSelect;
    matchReason: string[];
    openPostings: typeof jobPostings.$inferSelect[];
  }>;
  allFairChanceJobs: typeof jobPostings.$inferSelect[];
}> {
  // 1. Get learner's earned credentials
  const earnedCerts = await db
    .select({ levelTitle: certificates.levelTitle })
    .from(certificates)
    .where(eq(certificates.userId, userId));

  const credentialNames = earnedCerts.map((c) =>
    c.levelTitle.toLowerCase().replace(" trade certification", "").trim()
  );

  // 2. Get all active fair-chance employers
  const fairChanceEmployers = await db
    .select()
    .from(employerPartners)
    .where(
      and(
        eq(employerPartners.banTheBox, true),
        eq(employerPartners.partnershipStatus, "active")
      )
    );

  // 3. Get open barrier-friendly postings per employer
  const matches = await Promise.all(
    fairChanceEmployers.map(async (employer) => {
      const postings = await db
        .select()
        .from(jobPostings)
        .where(
          and(
            eq(jobPostings.employerId, employer.id),
            eq(jobPostings.status, "open")
          )
        );

      // Match reasons: credential overlap + barrier-friendly flags
      const matchReason: string[] = [];
      if (employer.banTheBox) matchReason.push("Ban-the-box employer");
      if (employer.fairChanceHiring) matchReason.push("Fair-chance hiring policy");
      if (employer.barrierFriendly) matchReason.push("Barrier-friendly workplace");

      // Check if any posting title matches a credential
      for (const posting of postings) {
        for (const cred of credentialNames) {
          if (
            posting.title.toLowerCase().includes(cred) ||
            (posting.requirements ?? "").toLowerCase().includes(cred)
          ) {
            matchReason.push(`Matches your ${cred} certification`);
            break;
          }
        }
      }

      return {
        employer,
        matchReason: [...new Set(matchReason)],
        openPostings: postings.filter((p) => p.status === "open"),
      };
    })
  );

  // 4. All open barrier-friendly jobs (for learner job board)
  const allFairChanceJobs = await db
    .select()
    .from(jobPostings)
    .where(
      and(
        eq(jobPostings.status, "open"),
        eq(jobPostings.barrierFriendly, true)
      )
    );

  return {
    userId,
    credentialsEarned: credentialNames,
    fairChanceMatches: matches.filter((m) => m.openPostings.length > 0),
    allFairChanceJobs,
  };
}

// FILE 4: server/lesson-mastery.ts
// Mastery gating at 80%, remediation branch, Spark trigger flag
// Modifies quiz submit handler — replaces 70% pass threshold
// ============================================================

export const MASTERY_THRESHOLD = 0.80; // 80% to advance
export const MAX_ATTEMPTS_BEFORE_REMEDIATION = 3;

export interface QuizResult {
  score: number;
  total: number;
  percentage: number;
  passed: boolean;
  attemptNumber: number;
  pointsEarned: number;
  masteryAchieved: boolean;
  triggerSpark: boolean;         // ← NEW: signal to frontend to auto-open Spark
  sparkContext: string;          // ← NEW: which questions to focus on
  remediation: {
    required: boolean;
    missedConcepts: string[];    // question titles/topics the learner missed
    suggestedReview: string;     // lesson or module to review
  };
  canAdvance: boolean;           // ← NEW: explicit gate
}

export async function processQuizSubmission(
  moduleId: string,
  userId: string,
  answers: Record<string, string>,
  questions: Array<{ id: string; correctAnswer: string; title?: string; topic?: string }>,
  attemptNumber: number
): Promise<QuizResult> {
  let correct = 0;
  const missedQuestions: Array<{ id: string; title?: string; topic?: string }> = [];

  for (const q of questions) {
    if (answers[q.id] === q.correctAnswer) {
      correct++;
    } else {
      missedQuestions.push(q);
    }
  }

  const percentage = questions.length > 0 ? correct / questions.length : 0;
  const masteryAchieved = percentage >= MASTERY_THRESHOLD;
  const needsRemediation =
    !masteryAchieved && attemptNumber >= MAX_ATTEMPTS_BEFORE_REMEDIATION;

  // Points: 100 for mastery, 50 for pass (70–79%), 25 for fail
  const pointsEarned = masteryAchieved ? 100 : percentage >= 0.70 ? 50 : 25;

  // Spark context — list the missed topics for the AI to focus on
  const missedConcepts = missedQuestions
    .map((q) => q.topic ?? q.title ?? `Question ${q.id}`)
    .filter(Boolean);

  const sparkContext =
    missedConcepts.length > 0
      ? `The learner just failed a quiz. They missed questions about: ${missedConcepts.join(", ")}. Help them understand these specific concepts. Do not just re-explain everything — focus only on what they got wrong.`
      : "";

  return {
    score: correct,
    total: questions.length,
    percentage,
    passed: percentage >= 0.70,       // still track "pass" at 70%
    attemptNumber,
    pointsEarned,
    masteryAchieved,                  // 80% = can advance
    triggerSpark: !masteryAchieved,   // auto-open Spark on any non-mastery result
    sparkContext,
    remediation: {
      required: needsRemediation,
      missedConcepts,
      suggestedReview: needsRemediation
        ? `Review the lesson content before attempting again. Focus on: ${missedConcepts.slice(0, 2).join(" and ")}.`
        : "",
    },
    canAdvance: masteryAchieved,      // hard gate — 80% required to advance
  };
}

/*
Replace the Spark section in client/src/pages/lesson-viewer.tsx
with this component. It:
  1. Auto-opens on quiz fail (triggerSpark prop)
  2. Injects missed-concept context into first message
  3. Sets grade level from learner profile (not user dropdown)
  4. Shows inline (not collapsible) when auto-triggered
  5. Reverts to collapsible when learner dismisses
*/

export const SparkAutoComponent = `
import { useState, useEffect, useRef } from "react";
import { Sparkles, X } from "lucide-react";

interface SparkAutoProps {
  lessonTitle: string;
  lessonContent: string;          // full content, not just 300 chars
  triggerSpark?: boolean;          // auto-open signal from quiz result
  sparkContext?: string;           // focused context from quiz failure
  gradeLevel?: "3-5" | "6-8" | "9-12" | "adult"; // from learner profile
  language?: "en" | "es";
}

export function SparkAuto({
  lessonTitle,
  lessonContent,
  triggerSpark = false,
  sparkContext,
  gradeLevel = "adult",
  language = "en",
}: SparkAutoProps) {
  const [isOpen, setIsOpen] = useState(false);
  const [autoTriggered, setAutoTriggered] = useState(false);
  const [messages, setMessages] = useState<Array<{ role: string; content: string }>>([]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const inputRef = useRef<HTMLInputElement>(null);

  // Auto-open when quiz signals failure
  useEffect(() => {
    if (triggerSpark && !autoTriggered) {
      setIsOpen(true);
      setAutoTriggered(true);
      // Inject a focused opening message from Spark
      if (sparkContext) {
        setMessages([{
          role: "assistant",
          content: language === "es"
            ? "Vi que tuviste algunas dificultades con el cuestionario. Déjame ayudarte con los conceptos específicos que necesitas repasar."
            : "I noticed you had some difficulty with the quiz. Let me help you work through the specific concepts you need to review.",
        }]);
        // Send the spark context as a silent system message
        sendSilentContext(sparkContext);
      }
    }
  }, [triggerSpark]);

  async function sendSilentContext(context: string) {
    // Prime the conversation without showing the system message
    try {
      await fetch("/api/ai-companion/chat", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          message: context,
          gradeLevel,
          lessonContext: lessonTitle + ": " + lessonContent.substring(0, 800),
          conversationHistory: [],
          language,
          isSystemPrime: true, // flag to backend to not return this in history
        }),
      });
    } catch { /* silent — don't surface errors from priming */ }
  }

  async function sendMessage() {
    if (!input.trim() || loading) return;
    const userMessage = input.trim();
    setInput("");
    setLoading(true);

    const updatedHistory = [...messages, { role: "user", content: userMessage }];
    setMessages(updatedHistory);

    try {
      const res = await fetch("/api/ai-companion/chat", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          message: userMessage,
          gradeLevel,
          lessonContext: lessonTitle + ": " + lessonContent.substring(0, 800),
          conversationHistory: messages,
          language,
        }),
      });
      const data = await res.json();
      setMessages([...updatedHistory, { role: "assistant", content: data.response ?? data.message ?? "" }]);
    } catch {
      setMessages([...updatedHistory, { role: "assistant", content: "I'm having trouble connecting right now. Try again in a moment." }]);
    } finally {
      setLoading(false);
      inputRef.current?.focus();
    }
  }

  // Collapsed state — learner can still open manually
  if (!isOpen) {
    return (
      <button
        onClick={() => setIsOpen(true)}
        className="w-full flex items-center gap-2 px-4 py-2 rounded-lg border border-purple-200 text-purple-700 hover:bg-purple-50 transition-colors text-sm font-medium"
      >
        <Sparkles className="h-4 w-4" />
        {language === "es" ? "¿Necesitas ayuda? Pregúntale a Spark" : "Need help? Ask Spark"}
      </button>
    );
  }

  // Open state
  return (
    <div className={\`
      flex flex-col rounded-xl border shadow-lg overflow-hidden
      \${autoTriggered && triggerSpark
        ? "border-orange-300 bg-orange-50"   // visual distinction when auto-triggered
        : "border-purple-200 bg-white"
      }
    \`} style={{ height: 420 }}>
      {/* Header */}
      <div className={\`
        flex items-center justify-between px-4 py-3
        \${autoTriggered && triggerSpark ? "bg-orange-100" : "bg-purple-50"}
      \`}>
        <div className="flex items-center gap-2">
          <Sparkles className={\`h-4 w-4 \${autoTriggered && triggerSpark ? "text-orange-600" : "text-purple-600"}\`} />
          <span className="text-sm font-semibold text-gray-700">
            Spark
            {autoTriggered && triggerSpark && (
              <span className="ml-2 text-xs font-normal text-orange-600">
                — Let's review what you missed
              </span>
            )}
          </span>
        </div>
        <button onClick={() => setIsOpen(false)} className="text-gray-400 hover:text-gray-600">
          <X className="h-4 w-4" />
        </button>
      </div>

      {/* Messages */}
      <div className="flex-1 overflow-y-auto p-4 space-y-3">
        {messages.length === 0 && (
          <p className="text-sm text-gray-500 text-center mt-8">
            {language === "es"
              ? "¡Hola! Soy Spark. ¿En qué puedo ayudarte hoy?"
              : "Hi! I'm Spark. What can I help you with today?"}
          </p>
        )}
        {messages.map((m, i) => (
          <div key={i} className={\`flex \${m.role === "user" ? "justify-end" : "justify-start"}\`}>
            <div className={\`
              max-w-xs rounded-lg px-3 py-2 text-sm
              \${m.role === "user"
                ? "bg-purple-600 text-white"
                : "bg-gray-100 text-gray-800"
              }
            \`}>
              {m.content}
            </div>
          </div>
        ))}
        {loading && (
          <div className="flex justify-start">
            <div className="bg-gray-100 rounded-lg px-3 py-2 text-sm text-gray-500">
              Spark is thinking…
            </div>
          </div>
        )}
      </div>

      {/* Input */}
      <div className="flex gap-2 px-4 py-3 border-t">
        <input
          ref={inputRef}
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === "Enter" && sendMessage()}
          placeholder={language === "es" ? "Escribe tu pregunta…" : "Ask Spark anything…"}
          className="flex-1 text-sm border border-gray-200 rounded-lg px-3 py-2 focus:outline-none focus:ring-2 focus:ring-purple-300"
        />
        <button
          onClick={sendMessage}
          disabled={loading || !input.trim()}
          className="px-4 py-2 bg-purple-600 text-white text-sm font-medium rounded-lg disabled:opacity-40 hover:bg-purple-700 transition-colors"
        >
          Send
        </button>
      </div>
    </div>
  );
}
`;
